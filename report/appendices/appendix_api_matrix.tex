\chapter{API Endpoint Matrix}

\begin{table}[H]
\centering
\scriptsize
\caption{Main API Endpoints and Responsibilities}
\begin{tabularx}{\textwidth}{L{0.25\textwidth} L{0.18\textwidth} X}
\toprule
\textbf{Endpoint} & \textbf{Method} & \textbf{Purpose} \\
\midrule
\code{/api/auth.php} & GET & Return authentication/session status and current user info. \\
\code{/api/auth.php?action=login} & POST & Authenticate and start session. \\
\code{/api/auth.php?action=logout} & POST & Terminate session. \\
\code{/api/projects.php} & GET / POST & List user projects and create project. \\
\code{/api/projects.php?id=...} & GET / PUT / DELETE & Read, update, or delete a specific project. \\
\code{/api/projects.php?code=...} & GET & Find project by invitation code. \\
\code{/api/members.php?project\_id=...} & GET & List project members. \\
\code{/api/members.php} & POST & Add member by username/email (default \code{action=add}). \\
\code{/api/members.php?action=leave} & POST & Leave a project as current member. \\
\code{/api/members.php?project\_id=...\&user\_id=...} & DELETE & Remove member from project (owner only). \\
\code{/api/join-requests.php} & POST & Submit join request. \\
\code{/api/join-requests.php?project\_id=...} & GET & List pending requests for owner. \\
\code{/api/join-requests.php?my=1} & GET & List current user's join-request history. \\
\code{/api/join-requests.php?id=...\&action=approve} & PUT & Approve request and add member. \\
\code{/api/join-requests.php?id=...\&action=reject} & PUT & Reject join request. \\
\code{/api/tasks.php?project\_id=...} & GET & List all tasks in a project. \\
\code{/api/tasks.php?project\_id=...\&search=...} & GET & Search/filter tasks with pagination metadata. \\
\code{/api/tasks.php?project\_id=...\&filters=1} & GET & Return filter options (members, status, priority). \\
\code{/api/tasks.php} & POST & Create task in project. \\
\code{/api/tasks.php?id=...} & GET / PUT / DELETE & Read, update, or delete task. \\
\code{/api/tasks.php?id=...\&action=status} & PUT & Update task status (drag and drop). \\
\code{/api/tasks.php?id=...\&action=claim} & PUT & Claim unassigned task for current member. \\
\code{/api/upload.php?task\_id=...} & POST / DELETE & Upload or remove task attachment. \\
\code{/api/upload.php?file=...} & GET & Download attachment file. \\
\bottomrule
\end{tabularx}
\end{table}
